```python
@dataclass(frozen=True)
class SourceTensor:
    name: str
    dtype: str
    shape: tuple[int, ...]
    offset: int
    nbytes: int

def output_name(architecture: str, source_name: str) -> str:
    prefix = {"qwen_drive_planner": "planning_expert.", "qwen_drive_perception": "bev_modeling."}[architecture]
    if not source_name.startswith(prefix):
        raise ValueError(f"unexpected tensor {source_name!r} for {architecture}")
    return f"{architecture}.{source_name[len(prefix):]}"

def payload_chunks(path: Path, offset: int, nbytes: int, chunk_size: int = 1 << 20):
    with path.open("rb") as handle:
        handle.seek(offset)
        remaining = nbytes
        while remaining:
            chunk = handle.read(min(chunk_size, remaining))
            if not chunk:
                raise ValueError(f"truncated tensor payload in {path}")
            remaining -= len(chunk)
            yield chunk
```

Use `GgufWriter.add_tensor_chunks` so the 9 GB VLM and 2 GB planners are never copied into one Python `bytes` object.
